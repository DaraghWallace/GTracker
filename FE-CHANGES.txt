  AFTER (also removes two console.logs, item 5.3, and fails loudly, item 5.8)
    export async function createSession(newSession: Omit<session, "sessionId">): Promise<session> {
        const token = await getToken()
        const url = `${API_URL}sessions`;

        const response = await fetch(url, {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
                "Authorization": token ? `Bearer ${token}` : "",
            },
            body: JSON.stringify(newSession),
        });

        const result = await response.json().catch(() => null);
        if (!response.ok) {
            throw new Error(result?.error ?? "Failed to create session");
        }
        return result as session;   // has the server-made sessionId, userId and dateDone
    }

3. Read the role from the GROUPS, not the attribute                                           (item 5.1, 1.1)
---------------------------------------------------------------------------------------------------------
New accounts have no custom:userType any more, so the trainer view and the member view would show nothing.
Helpers/amplify.ts : getUserAttributes()

  BEFORE
    return {
      userId: payload?.sub,
      email: payload?.email,
      nickname: payload?.nickname,
      userType: payload?.["custom:userType"],
    };

  AFTER
    // Roles come from the Cognito groups in the token. This only decides what the UI SHOWS; the API
    // checks the same groups on the server.
    const groups = (payload?.["cognito:groups"] as string[] | undefined) ?? [];
    const userType = groups.includes("admins") ? "developer"
                   : groups.includes("trainers") ? "trainer"
                   : "member";
    return {
      userId: payload?.sub,
      email: payload?.email,
      nickname: payload?.nickname,
      userType,
    };

  Nothing else changes: Header.tsx, Coaching.tsx and App.tsx keep switching on "developer" / "trainer" / "member".
  Your own admin account must be in the "admins" group (it is, you saw it in the token), and every real trainer
  must be in "trainers". Groups only appear in a token issued AFTER the user was added, so log out and in again.
